cmake_minimum_required(VERSION 3.22)
project(OpenDiag VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
option(OPENDIAG_WORKER_ONLY "Build the standalone J2534 driver host without Qt" OFF)
option(OPENDIAG_LEGACY_WIN32 "Also build the original Win32 UI reconstruction" OFF)
option(OPENDIAG_DEPLOY_QT "Bundle Qt in a macOS or Windows installation" OFF)

add_library(diagnostics STATIC native/diagnostics.cpp)
add_executable(opendiag-driver native/worker.cpp native/replay.cpp)
target_link_libraries(opendiag-driver PRIVATE diagnostics ${CMAKE_DL_LIBS})
if(MINGW)
    # The x86 host may live beside x64 DLLs; do not depend on a MinGW runtime DLL.
    target_link_options(opendiag-driver PRIVATE -municode -static)
endif()
if(WIN32 AND CMAKE_SIZEOF_VOID_P EQUAL 4)
    set_target_properties(opendiag-driver PROPERTIES OUTPUT_NAME opendiag-driver-x86)
endif()
if(OPENDIAG_WORKER_ONLY)
    if(MSVC)
        set_target_properties(opendiag-driver diagnostics PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    install(TARGETS opendiag-driver RUNTIME DESTINATION bin)
    if(BUILD_TESTING)
        add_library(opendiag-sim SHARED native/simulator.cpp)
        find_package(Python3 REQUIRED COMPONENTS Interpreter)
        add_test(NAME worker-simulator COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/session.py $<TARGET_FILE:opendiag-driver> $<TARGET_FILE:opendiag-sim>)
    endif()
    return()
endif()

find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets)
add_library(opendiag-sim SHARED native/simulator.cpp)
add_library(images STATIC native/image.cpp native/files.cpp native/virtual_ecu.cpp)
target_link_libraries(images PUBLIC Qt6::Core)
set_target_properties(images PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_definitions(opendiag-sim PRIVATE OPENDIAG_VIRTUAL_ECU)
target_link_libraries(opendiag-sim PRIVATE images)
add_executable(opendiag-image native/image_cli.cpp)
target_link_libraries(opendiag-image PRIVATE images)
add_library(desktop STATIC native/window.cpp native/monitor.cpp native/acquisition.cpp native/viewer.cpp native/can_monitor.cpp native/definitions.cpp native/program_dialog.cpp)
target_compile_definitions(desktop PRIVATE OPENDIAG_SIM_FILE="$<TARGET_FILE_NAME:opendiag-sim>")
target_link_libraries(desktop PUBLIC Qt6::Widgets images diagnostics)
qt_add_executable(OpenDiag WIN32 MACOSX_BUNDLE native/main.cpp)
target_link_libraries(OpenDiag PRIVATE desktop)
add_dependencies(OpenDiag opendiag-driver opendiag-sim)
set_target_properties(OpenDiag PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER org.opendiag.OpenDiag)
if(APPLE)
    set(tools_dir OpenDiag.app/Contents/MacOS)
    set_target_properties(opendiag-image PROPERTIES INSTALL_RPATH "@executable_path/../Frameworks")
    # Keep the build-tree .app launchable before installation as well.
    add_custom_target(app-tools ALL
        COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:opendiag-driver> $<TARGET_FILE:opendiag-sim> $<TARGET_FILE:opendiag-image> $<TARGET_FILE_DIR:OpenDiag>
        COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/reference/examples" "$<TARGET_BUNDLE_CONTENT_DIR:OpenDiag>/Resources/examples"
        DEPENDS OpenDiag opendiag-driver opendiag-sim opendiag-image)
else()
    set(tools_dir bin)
    add_custom_command(TARGET OpenDiag POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/reference/examples" "$<TARGET_FILE_DIR:OpenDiag>/examples")
endif()
install(TARGETS OpenDiag BUNDLE DESTINATION . RUNTIME DESTINATION bin)
install(TARGETS opendiag-driver opendiag-sim opendiag-image RUNTIME DESTINATION ${tools_dir} LIBRARY DESTINATION ${tools_dir})
if(APPLE)
    install(DIRECTORY reference/examples DESTINATION OpenDiag.app/Contents/Resources)
elseif(WIN32)
    install(DIRECTORY reference/examples DESTINATION .)
else()
    install(DIRECTORY reference/examples DESTINATION share/opendiag)
endif()
if(WIN32)
    set(OPENDIAG_X86_WORKER "" CACHE FILEPATH "Built x86 worker to include in the x64 installer")
    if(OPENDIAG_X86_WORKER)
        install(PROGRAMS "${OPENDIAG_X86_WORKER}" DESTINATION bin)
    endif()
endif()
if(OPENDIAG_DEPLOY_QT AND APPLE)
    find_program(MACDEPLOYQT macdeployqt HINTS "${QT6_INSTALL_PREFIX}/bin" REQUIRED)
    install(FILES $<TARGET_FILE:Qt6::QCocoaIntegrationPlugin> DESTINATION OpenDiag.app/Contents/PlugIns/platforms)
    install(FILES $<TARGET_FILE:Qt6::QMacStylePlugin> DESTINATION OpenDiag.app/Contents/PlugIns/styles)
    install(FILES native/qt.conf DESTINATION OpenDiag.app/Contents/Resources)
    install(CODE "set(app \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/OpenDiag.app\")
        execute_process(COMMAND \"${MACDEPLOYQT}\" \"\${app}\" -no-plugins \"-executable=\${app}/Contents/MacOS/opendiag-image\" \"-executable=\${app}/Contents/MacOS/$<TARGET_FILE_NAME:opendiag-sim>\" \"-executable=\${app}/Contents/PlugIns/platforms/$<TARGET_FILE_NAME:Qt6::QCocoaIntegrationPlugin>\" \"-executable=\${app}/Contents/PlugIns/styles/$<TARGET_FILE_NAME:Qt6::QMacStylePlugin>\" -always-overwrite COMMAND_ERROR_IS_FATAL ANY)
        execute_process(COMMAND /usr/bin/codesign --force --deep --sign - \"\${app}\" COMMAND_ERROR_IS_FATAL ANY)")
elseif(OPENDIAG_DEPLOY_QT AND WIN32)
    find_program(WINDEPLOYQT windeployqt HINTS "${QT6_INSTALL_PREFIX}/bin" REQUIRED)
    install(CODE "execute_process(COMMAND \"${WINDEPLOYQT}\" --release --no-translations \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/bin/OpenDiag.exe\" COMMAND_ERROR_IS_FATAL ANY)")
endif()

if(OPENDIAG_LEGACY_WIN32 AND WIN32)
    enable_language(C RC)
    add_executable(ptdiag_reversed WIN32 dummy_callbacks.c utils.c configuration.c init_file.c about.c main_dialog.c logfile.c ptdiag.c ptdiag.rc)
endif()
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()

set(CPACK_PACKAGE_VENDOR "OpenDiag")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Ford diagnostics, firmware tools and virtual ECU programming and recovery")
set(CPACK_PACKAGE_DIRECTORY "${CMAKE_BINARY_DIR}/packages")
if(APPLE)
    set(CPACK_GENERATOR DragNDrop)
elseif(WIN32)
    set(CPACK_GENERATOR NSIS)
    set(CPACK_NSIS_MENU_LINKS "bin/OpenDiag.exe" "OpenDiag")
else()
    install(FILES native/org.opendiag.OpenDiag.desktop DESTINATION share/applications)
    set(CPACK_GENERATOR DEB)
    set(CPACK_DEBIAN_PACKAGE_MAINTAINER "OpenDiag contributors")
    set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
    set(CPACK_DEBIAN_PACKAGE_DEPENDS "qt6-qpa-plugins")
endif()
include(CPack)
