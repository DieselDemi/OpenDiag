find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Qt6 6.4 REQUIRED COMPONENTS Test)
add_executable(opendiag-ui-test ui.cpp)
target_link_libraries(opendiag-ui-test PRIVATE desktop Qt6::Test)
set_target_properties(opendiag-ui-test PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
add_dependencies(opendiag-ui-test opendiag-driver opendiag-sim)
add_test(NAME desktop-session COMMAND opendiag-ui-test)
set_tests_properties(desktop-session PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 20)
add_test(NAME simulator-session COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/session.py $<TARGET_FILE:opendiag-driver> $<TARGET_FILE:opendiag-sim>)
add_test(NAME image-files COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/images.py $<TARGET_FILE:opendiag-image>)
add_library(fault-provider SHARED fault_provider.cpp)
add_test(NAME driver-errors COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/faults.py $<TARGET_FILE:opendiag-driver> $<TARGET_FILE:fault-provider>)
